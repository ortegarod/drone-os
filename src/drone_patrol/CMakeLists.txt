cmake_minimum_required(VERSION 3.8)
project(drone_patrol)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(px4_ros2_cpp REQUIRED)
find_package(drone_interfaces REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
add_executable(patrol_node src/patrol_node.cpp)
target_include_directories(patrol_node PUBLIC include)
ament_target_dependencies(patrol_node rclcpp px4_msgs px4_ros2_cpp drone_interfaces)
target_link_libraries(patrol_node Eigen3::Eigen)
install(TARGETS patrol_node DESTINATION lib/${PROJECT_NAME})
ament_package()
